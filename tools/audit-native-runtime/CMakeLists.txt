cmake_minimum_required(VERSION 3.20)
project(AWE64NativeRuntimeFaultAudit LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows-native audit only")
endif()
add_library(AWE64DOSBoxRuntime SHARED fault_dll.cpp)
set_target_properties(AWE64DOSBoxRuntime PROPERTIES OUTPUT_NAME Engine)
target_include_directories(AWE64DOSBoxRuntime PRIVATE ../../integration)
target_compile_features(AWE64DOSBoxRuntime PRIVATE cxx_std_20)
add_executable(native_runtime_fault_audit host.cpp)
target_include_directories(native_runtime_fault_audit PRIVATE ../../src ../../integration)
target_compile_features(native_runtime_fault_audit PRIVATE cxx_std_20)
target_link_libraries(native_runtime_fault_audit PRIVATE bcrypt)
add_dependencies(native_runtime_fault_audit AWE64DOSBoxRuntime)
